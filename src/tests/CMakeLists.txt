# Only the C++ unit tests are wired into CTest.
#
# The rest of the automake suite (local.test, the emulation-*.test family, the
# e2e tests) is POSIX shell driving tmux and a pty, so it stays under
# `make check` on POSIX and has no meaning on Windows. These four, by contrast,
# are pure computation and are exactly the regression net we want available on
# the Windows port.

add_executable(test-ocb-aes ocb-aes.cc test_utils.cc test_utils.h)
target_link_libraries(test-ocb-aes PRIVATE mosh_common moshcrypto moshutil)

add_executable(test-encrypt-decrypt encrypt-decrypt.cc test_utils.cc test_utils.h)
target_link_libraries(test-encrypt-decrypt PRIVATE mosh_common moshcrypto moshutil)

add_executable(test-nonce-incr nonce-incr.cc)
target_link_libraries(test-nonce-incr PRIVATE mosh_common moshnetwork moshcrypto moshutil)

add_executable(test-utf8 utf8.cc)
target_link_libraries(test-utf8 PRIVATE mosh_common moshutil_portable)
add_test(NAME utf8 COMMAND test-utf8)

add_executable(test-uniwidth uniwidth.cc)
target_link_libraries(test-uniwidth PRIVATE mosh_common moshutil_portable)
add_test(NAME uniwidth COMMAND test-uniwidth)

# The end-to-end character path. Portable on purpose: this is the test that
# catches the 16-bit wchar_t bug, and it has to be runnable on Windows.
add_executable(test-unicode-roundtrip unicode-roundtrip.cc)
target_link_libraries(test-unicode-roundtrip PRIVATE mosh_common moshterminal moshutil_portable)
add_test(NAME unicode-roundtrip COMMAND test-unicode-roundtrip)

add_test(NAME ocb-aes         COMMAND test-ocb-aes)
add_test(NAME encrypt-decrypt COMMAND test-encrypt-decrypt)
add_test(NAME nonce-incr      COMMAND test-nonce-incr)

# base64's vector table is generated by a Perl script. Perl is a given on
# POSIX but not on Windows, so this test is opt-in on availability rather than
# a hard build dependency.
find_package(Perl QUIET)
if(PERL_FOUND)
  set(_b64 "${CMAKE_CURRENT_BINARY_DIR}/base64_vector.cc")
  add_custom_command(
    OUTPUT "${_b64}"
    COMMAND ${CMAKE_COMMAND} -E echo "#include \"base64_vector.h\"" > "${_b64}"
    COMMAND ${PERL_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/genbase64.pl" >> "${_b64}"
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/genbase64.pl"
    COMMENT "Generating base64 test vectors"
    VERBATIM)
  add_executable(test-base64 base64.cc "${_b64}" base64_vector.h)
  target_include_directories(test-base64 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(test-base64 PRIVATE mosh_common moshcrypto moshutil)
  add_test(NAME base64 COMMAND test-base64)
else()
  message(STATUS "  base64 test   : skipped (no perl)")
endif()
